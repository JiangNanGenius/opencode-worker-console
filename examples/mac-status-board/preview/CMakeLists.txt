cmake_minimum_required(VERSION 3.16)
project(StatusBoardPreview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LGFX "${CMAKE_CURRENT_SOURCE_DIR}/../.pio/libdeps/info-board/LovyanGFX/src")
file(GLOB LIBSRC ${LGFX}/lgfx/v1/*.cpp ${LGFX}/lgfx/v1/misc/*.cpp ${LGFX}/lgfx/v1/platforms/sdl/*.cpp ${LGFX}/lgfx/utility/*.c)
list(APPEND LIBSRC ${LGFX}/lgfx/v1/panel/Panel_Device.cpp ${LGFX}/lgfx/v1/panel/Panel_FrameBufferBase.cpp ${LGFX}/lgfx/Fonts/efont/lgfx_efont_cn.c)
find_package(SDL2 REQUIRED)
add_executable(board_preview main.cpp ${LIBSRC})
target_compile_definitions(board_preview PRIVATE LGFX_SDL LGFX_USE_V1)
target_compile_options(board_preview PRIVATE -ffunction-sections -fdata-sections)
target_link_options(board_preview PRIVATE -Wl,-dead_strip)
target_include_directories(board_preview PRIVATE ${LGFX} ../include ../.pio/libdeps/info-board/ArduinoJson/src)
target_link_libraries(board_preview SDL2::SDL2)
add_executable(board_tests ../tests/model_tests.cpp)
target_include_directories(board_tests PRIVATE ../include ../.pio/libdeps/info-board/ArduinoJson/src)

target_compile_options(board_tests PRIVATE -UNDEBUG)
